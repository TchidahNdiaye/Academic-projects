/*====================================================================
  BIBLIOTHEQUE DAX - 4 PAGES DATA QUALITY + STAGE 3 IFRS 9
  Tables de faits recommandees :
    F_DQ_SCORE, F_DQ_DIMENSIONS, F_DQ_RESULTATS,
    F_STAGE3_STOCKS, F_STAGE3_PONTS

  Les 2 segments Stage 3 sont :
    DIM_SCOPE_MODELE[SCOPE_MODELE] : TOUS, CAV, CCO, CDI, COR, CRN, IMO
    DIM_SCOPE_RISQUE[SCOPE_RISQUE] : TOUS, DOUTEUX, COMPROMIS, NON_RENSEIGNE

  Les 2 dimensions de scope sont reliees a F_STAGE3_PONTS, mais restent
  deconnectees de F_STAGE3_STOCKS. Les mesures de stock lisent la selection
  et l'appliquent aux dimensions reelles DIM_MODELE et DIM_NATURE_RISQUE.
====================================================================*/

/*----------------- TABLES DE REFERENCE A CREER EN DAX -------------------*/

DIM_SCOPE_MODELE =
DATATABLE(
    "SCOPE_MODELE", STRING,
    {
        {"TOUS"}, {"CAV"}, {"CCO"}, {"CDI"}, {"COR"}, {"CRN"}, {"IMO"}
    }
)

DIM_SCOPE_RISQUE =
DATATABLE(
    "SCOPE_RISQUE", STRING,
    {
        {"TOUS"}, {"DOUTEUX"}, {"COMPROMIS"}, {"NON_RENSEIGNE"}
    }
)

DIM_ETAPE_PONT =
DATATABLE(
    "ORDRE_ETAPE", INTEGER,
    "ETAPE", STRING,
    {
        {1, "Stock M-1"},
        {2, "Sorties perimetre"},
        {3, "Entrees perimetre"},
        {4, "Variation communs"},
        {5, "Stock M"}
    }
)

/* DIM_MODELE : CAV, CCO, CDI, COR, CRN, IMO.
   DIM_NATURE_RISQUE : DOUTEUX, COMPROMIS, NON_RENSEIGNE.
   DIM_ARRETE : une ligne unique par ARRETE, avec ROLE = M ou M-1.
   Construire ces 3 dimensions dans Power Query a partir des distincts des
   tables, ou en DAX si les listes sont strictement maitrisees. */

/*----------------- RELATIONS A CREER DANS LE MODELE ---------------------*/
/* 1:* sens unique DIM -> FACT, aucune relation fait-a-fait.

   DIM_ARRETE[ARRETE] -> F_STAGE3_STOCKS[ARRETE]
   DIM_ARRETE[ARRETE] -> F_DQ_SCORE[arrete]
   DIM_ARRETE[ARRETE] -> F_DQ_DIMENSIONS[arrete]
   DIM_ARRETE[ARRETE] -> F_DQ_RESULTATS[arrete]

   DIM_MODELE[MODEL] -> F_STAGE3_STOCKS[MODEL]
   DIM_MODELE[MODEL] -> F_DQ_DIMENSIONS[model]
   DIM_MODELE[MODEL] -> F_DQ_RESULTATS[model]

   DIM_NATURE_RISQUE[NATURE_RISQUE_STD] -> F_STAGE3_STOCKS[NATURE_RISQUE_STD]

   DIM_CDE[cde_no] -> F_DQ_SCORE[cde_no]
   DIM_CDE[cde_no] -> F_DQ_DIMENSIONS[cde_no]
   DIM_CDE[cde_no] -> F_DQ_RESULTATS[cde_no]

   DIM_DIMENSION[dimension] -> F_DQ_DIMENSIONS[dimension]
   DIM_DIMENSION[dimension] -> F_DQ_RESULTATS[dimension]
   DIM_SEVERITE[severite] -> F_DQ_DIMENSIONS[severite]
   DIM_SEVERITE[severite] -> F_DQ_RESULTATS[severite]

   DIM_SCOPE_MODELE[SCOPE_MODELE] -> F_STAGE3_PONTS[SCOPE_MODELE]
   DIM_SCOPE_RISQUE[SCOPE_RISQUE] -> F_STAGE3_PONTS[SCOPE_RISQUE]
   DIM_ETAPE_PONT[ETAPE] -> F_STAGE3_PONTS[ETAPE]

   DIM_SCOPE_MODELE et DIM_SCOPE_RISQUE ne sont PAS reliees a F_STAGE3_STOCKS.
   Les mesures ci-dessous transmettent leur choix aux dimensions de stock.
*/

/*----------------- MOTIF COMMUN DE FILTRAGE DU STOCK --------------------*/
/* Dans chaque mesure de stock :
   - le contexte de ligne DIM_MODELE reste preserve (tableau / donut);
   - le segment Scope modele restreint ce contexte si un modele est choisi;
   - le segment Scope risque filtre DIM_NATURE_RISQUE.
*/

/*----------------- 1. CONTRATS ------------------------------------------*/
Nb contrats M =
VAR _SM = SELECTEDVALUE(DIM_SCOPE_MODELE[SCOPE_MODELE], "TOUS")
VAR _SR = SELECTEDVALUE(DIM_SCOPE_RISQUE[SCOPE_RISQUE], "TOUS")
VAR _Modeles = FILTER(VALUES(DIM_MODELE[MODEL]), _SM = "TOUS" || DIM_MODELE[MODEL] = _SM)
VAR _Risques = FILTER(VALUES(DIM_NATURE_RISQUE[NATURE_RISQUE_STD]), _SR = "TOUS" || DIM_NATURE_RISQUE[NATURE_RISQUE_STD] = _SR)
RETURN
CALCULATE(
    SUM(F_STAGE3_STOCKS[NB_CONTRATS]),
    DIM_ARRETE[ROLE] = "M",
    _Modeles,
    _Risques
)

Nb contrats M-1 =
VAR _SM = SELECTEDVALUE(DIM_SCOPE_MODELE[SCOPE_MODELE], "TOUS")
VAR _SR = SELECTEDVALUE(DIM_SCOPE_RISQUE[SCOPE_RISQUE], "TOUS")
VAR _Modeles = FILTER(VALUES(DIM_MODELE[MODEL]), _SM = "TOUS" || DIM_MODELE[MODEL] = _SM)
VAR _Risques = FILTER(VALUES(DIM_NATURE_RISQUE[NATURE_RISQUE_STD]), _SR = "TOUS" || DIM_NATURE_RISQUE[NATURE_RISQUE_STD] = _SR)
RETURN
CALCULATE(
    SUM(F_STAGE3_STOCKS[NB_CONTRATS]),
    DIM_ARRETE[ROLE] = "M-1",
    _Modeles,
    _Risques
)

Variation contrats = [Nb contrats M] - [Nb contrats M-1]

Variation contrats (%) =
DIVIDE([Variation contrats], [Nb contrats M-1])

VAR _SM = SELECTEDVALUE(DIM_SCOPE_MODELE[SCOPE_MODELE], "TOUS")
VAR _SR = SELECTEDVALUE(DIM_SCOPE_RISQUE[SCOPE_RISQUE], "TOUS")
VAR _Modeles = FILTER(VALUES(DIM_MODELE[MODEL]), _SM = "TOUS" || DIM_MODELE[MODEL] = _SM)
VAR _Risques = FILTER(VALUES(DIM_NATURE_RISQUE[NATURE_RISQUE_STD]), _SR = "TOUS" || DIM_NATURE_RISQUE[NATURE_RISQUE_STD] = _SR)
RETURN
CALCULATE(
    SUM(F_STAGE3_STOCKS[EAD_EUR]),
    DIM_ARRETE[ROLE] = "M",
    _Modeles,
    _Risques
)

EAD M-1 (EUR) =
VAR _SM = SELECTEDVALUE(DIM_SCOPE_MODELE[SCOPE_MODELE], "TOUS")
VAR _SR = SELECTEDVALUE(DIM_SCOPE_RISQUE[SCOPE_RISQUE], "TOUS")
VAR _Modeles = FILTER(VALUES(DIM_MODELE[MODEL]), _SM = "TOUS" || DIM_MODELE[MODEL] = _SM)
VAR _Risques = FILTER(VALUES(DIM_NATURE_RISQUE[NATURE_RISQUE_STD]), _SR = "TOUS" || DIM_NATURE_RISQUE[NATURE_RISQUE_STD] = _SR)
RETURN
CALCULATE(
    SUM(F_STAGE3_STOCKS[EAD_EUR]),
    DIM_ARRETE[ROLE] = "M-1",
    _Modeles,
    _Risques
)

EAD M (M€) = DIVIDE([EAD M (EUR)], 1000000)
EAD M-1 (M€) = DIVIDE([EAD M-1 (EUR)], 1000000)
Variation EAD (EUR) = [EAD M (EUR)] - [EAD M-1 (EUR)]
Variation EAD (M€) = DIVIDE([Variation EAD (EUR)], 1000000)
Variation EAD (%) = DIVIDE([Variation EAD (EUR)], [EAD M-1 (EUR)])

/*----------------- 3. PROVISIONS / COUT DU RISQUE -----------------------*/

VAR _SM = SELECTEDVALUE(DIM_SCOPE_MODELE[SCOPE_MODELE], "TOUS")
VAR _SR = SELECTEDVALUE(DIM_SCOPE_RISQUE[SCOPE_RISQUE], "TOUS")
VAR _Modeles = FILTER(VALUES(DIM_MODELE[MODEL]), _SM = "TOUS" || DIM_MODELE[MODEL] = _SM)

CALCULATE(
    SUM(F_STAGE3_STOCKS[PROVISION_EUR]),
    DIM_ARRETE[ROLE] = "M",
    _Modeles,
    _Risques
)

Provision M-1 (EUR) =
VAR _SM = SELECTEDVALUE(DIM_SCOPE_MODELE[SCOPE_MODELE], "TOUS")
VAR _SR = SELECTEDVALUE(DIM_SCOPE_RISQUE[SCOPE_RISQUE], "TOUS")
VAR _Modeles = FILTER(VALUES(DIM_MODELE[MODEL]), _SM = "TOUS" || DIM_MODELE[MODEL] = _SM)

Provision M (M€) = DIVIDE([Provision M (EUR)], 1000000)
Provision M-1 (M€) = DIVIDE([Provision M-1 (EUR)], 1000000)
Variation provision (EUR) = [Provision M (EUR)] - [Provision M-1 (EUR)]
Variation provision (M€) = DIVIDE([Variation provision (EUR)], 1000000)
Variation provision (%) = DIVIDE([Variation provision (EUR)], [Provision M-1 (EUR)])

/* Convention metier demandee : le KPI CoR est ici la variation du stock de
   provisions M - M-1. Il doit etre libelle "Variation du stock de provisions
   (indicateur CoR retenu)" et non presente comme un P&L CoR exhaustif. */
Cout du risque retenu (M€) = [Variation provision (M€)]

/*----------------- 4. TAUX DE COUVERTURE COMPLEMENTAIRE -----------------*/
Taux de couverture M = DIVIDE([Provision M (EUR)], [EAD M (EUR)])
Taux de couverture M-1 = DIVIDE([Provision M-1 (EUR)], [EAD M-1 (EUR)])
Variation couverture (pt) = ([Taux de couverture M] - [Taux de couverture M-1]) * 100

/*----------------- 5. WATERFALLS ----------------------------------------*/
/* F_STAGE3_PONTS porte deja les signes : sorties negatives, entrees
   positives, delta communs signe M - M-1. */
Pont EAD (M€) =
DIVIDE(SUM(F_STAGE3_PONTS[EAD_PONT_EUR]), 1000000)

Pont provisions (M€) =
DIVIDE(SUM(F_STAGE3_PONTS[PROVISION_PONT_EUR]), 1000000)

Nb contrats mouvement = SUM(F_STAGE3_PONTS[NB_CONTRATS])

/* Waterfall : axe DIM_ETAPE_PONT[ETAPE], valeurs [Pont EAD (M€)] puis
   [Pont provisions (M€)]. Definir Stock M-1 et Stock M comme totaux. */

/*----------------- 6. PART DES SIX MODELES ------------------------------*/
Part EAD M par modele (%) =
DIVIDE(
    [EAD M (EUR)],
    CALCULATE([EAD M (EUR)], REMOVEFILTERS(DIM_MODELE[MODEL]))
)

Part provisions M par modele (%) =
DIVIDE(
    [Provision M (EUR)],
    CALCULATE([Provision M (EUR)], REMOVEFILTERS(DIM_MODELE[MODEL]))
)

/*----------------- 7. CONTROLES DATA QUALITY STAGE 3 --------------------*/

VAR _SM = SELECTEDVALUE(DIM_SCOPE_MODELE[SCOPE_MODELE], "TOUS")
VAR _SR = SELECTEDVALUE(DIM_SCOPE_RISQUE[SCOPE_RISQUE], "TOUS")

NB provisions manquantes M =

NB LGDD manquantes M =
VAR _SM = SELECTEDVALUE(DIM_SCOPE_MODELE[SCOPE_MODELE], "TOUS")

RETURN CALCULATE(SUM(F_STAGE3_STOCKS[NB_LGDD_MANQ]), DIM_ARRETE[ROLE] = "M", _Modeles, _Risques)

Taux completude EAD M =
1 - DIVIDE([NB EAD manquantes M], [Nb contrats M])

Taux completude provisions M =
1 - DIVIDE([NB provisions manquantes M], [Nb contrats M])

Taux completude LGDD M =
1 - DIVIDE([NB LGDD manquantes M], [Nb contrats M])

Statut DQ Stage 3 =
IF(
    [NB EAD manquantes M] + [NB provisions manquantes M] + [NB LGDD manquantes M] = 0,
    "OK sur les controles de completude affiches",
    "A CONTROLER - valeurs manquantes"
)

/*----------------- 8. MESURES MINIMALES DES 3 PAGES DQ EXISTANTES --------*/
/* D1 : le score DQ est deja calcule par CDE dans Excel/SAS. Le poids doit
   etre applique une fois, au niveau global; ne pas faire AVERAGE(score). */
Score DQ M =
DIVIDE(
    SUMX(FILTER(F_DQ_SCORE, F_DQ_SCORE[role] = "M"), F_DQ_SCORE[score] * F_DQ_SCORE[poids]),
    SUMX(FILTER(F_DQ_SCORE, F_DQ_SCORE[role] = "M"), F_DQ_SCORE[poids])
)

Score DQ M-1 =
DIVIDE(
    SUMX(FILTER(F_DQ_SCORE, F_DQ_SCORE[role] = "M-1"), F_DQ_SCORE[score] * F_DQ_SCORE[poids]),
    SUMX(FILTER(F_DQ_SCORE, F_DQ_SCORE[role] = "M-1"), F_DQ_SCORE[poids])
)

Ecart score DQ (pt) = ([Score DQ M] - [Score DQ M-1]) * 100

/* D2 : taux de controle = nbre_ok / nbre_total. Verifier que chaque ligne
   de F_DQ_DIMENSIONS represente bien une population non dupliquee au grain
   CDE x modele x dimension x arrete avant d'utiliser cette somme. */
Taux conformite dimension =
DIVIDE(
    SUM(F_DQ_DIMENSIONS[nbre_ok]),
    SUM(F_DQ_DIMENSIONS[nbre_total])
)

/* D4 : indicateurs disponibles dans les captures de Resultats_DQ. */
Anomalies DQ (nombre) = SUM(F_DQ_RESULTATS[nbre_ko_ano])
Exposition KO DQ (unite source) = SUM(F_DQ_RESULTATS[expo_ko_ano])

/* ATTENTION : les captures ne montrent pas de champ CAUSE_RACINE, STATUT
   nouvelle/persistante/resolue ni RULE_ID dans ces extractions. Pour D4,
   construire les visuels a partir de DIMENSION, VARIABLE, DESCRIPTION,
   SEVERITE, MODEL et expo_ko_ano disponibles, sans inventer un cycle de vie. */

/* Variations % : 0.0%; variations de couverture : 0.0 "pt" */
/* Les valeurs manquantes restent BLANK; ne pas remplacer par zero. */
