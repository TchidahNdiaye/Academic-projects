/* ===================================================== */
/* MACRO : CONVERSION EXPLICITE UTF-8 VERS LATIN1       */
/* ===================================================== */

%macro UTF8_TO_LATIN1(src=, dst=, audit=);

    %local _z_lib _z_mem _z_charlist;

    /* Identification de la table source */

    %let _z_lib=%upcase(%scan(&src,1,.));
    %let _z_mem=%upcase(%scan(&src,2,.));

    /* Récupération automatique des variables caractères */

    proc sql noprint;

        select name
        into :_z_charlist separated by ' '

        from dictionary.columns

        where libname="&_z_lib"
          and memname="&_z_mem"
          and type='char'

        order by varnum;

    quit;


    /* Vérification de la liste des variables */

    %if %superq(_z_charlist)= %then %do;

        %put ERROR: Aucune variable caractere identifiee.;
        %return;

    %end;


    /* ================================================= */
    /* LECTURE SANS TRANSCODAGE AUTOMATIQUE              */
    /* ================================================= */

    data &dst
         (drop=_z_i _z_before _z_latin
               _z_back _z_obs _z_var)

         &audit
         (keep=_z_obs _z_var);

        length _z_before _z_latin _z_back $32767;
        length _z_var $32 _z_obs 8;

        /* Lecture des octets UTF-8 sans conversion CEDA */

        set &src(encoding=any);

        /* Ensemble des variables caractères de la source */

        array _z_chars {*} &_z_charlist;

        do _z_i=1 to dim(_z_chars);

            /* Valeur originale */

            _z_before=_z_chars[_z_i];

            /* Conversion UTF-8 vers LATIN1 */

            _z_latin=kcvt(
                _z_before,
                'UTF-8',
                'LATIN1'
            );

            /* Conversion inverse pour contrôler les pertes */

            _z_back=kcvt(
                _z_latin,
                'LATIN1',
                'UTF-8'
            );

            /* Identification des conversions non réversibles */

            if strip(_z_back) ne strip(_z_before) then do;

                _z_obs=_n_;
                _z_var=vname(_z_chars[_z_i]);

                output &audit;

            end;

            /* Remplacement par la valeur convertie */

            _z_chars[_z_i]=_z_latin;

        end;

        /* Ecriture dans la bibliothèque LATIN1 */

        output &dst;

    run;

%mend;
